# @description Print the UTC time, `YYYY-MM-DDTHH:MM:SSZ` for a log line or `YYYYMMDD-HHMMSS` for a slot name, from

utc_stamp() {

# @description Set `slot_present` to 1 when the path `$1` exists (a symbolic link counted by itself) and to 0 when it
#   is absent, from one `/usr/bin/stat` whose status and message are read (regime-cutover U272g, standing order 21):
#   only the documented ENOENT is an absence; every other failure refuses (exit 2) naming the path and the message.
# @arg $1 string An absolute path.
# @set slot_present 1 or 0.
path_presence() {
    local err
    if err=$(/usr/bin/stat "${lstat_args[@]}" -- "$1" 2>&1 > /dev/null); then
        slot_present=1
        return 0
    fi
    slot_present=0
    case ${err} in
    *'No such file or directory'*) return 0 ;;
    esac
    echo "acp-broker: refused: cannot read whether ${1} exists (${err}); it may remain: remove it by hand" >&2
    exit 2
}

# @description Remove this connection's slot when it was made (the sandbox directory with its settings and canary).
#   A directory left without its owner's bits keeps rm out, so the mode is reset first; the removal fails closed: a
#   slot that remains is named and the function fails, so the connection never ends 0 with a slot behind.
# @exitcode 0 The slot is gone, or none was made.
# @exitcode 1 The slot remains, named on stderr.
slot_remove() {
    ((slot_made)) || return 0
    path_presence "${slot}"
    ((slot_present)) || return 0
    # Links in the tree are not followed (the traversal default of both chmods); a failure here shows below.
    if /bin/chmod -R u+rwX "${slot}"; then :; fi
    if /bin/rm -rf -- "${slot}"; then :; fi
    path_presence "${slot}"
    ((! slot_present)) || {
        echo "acp-broker: the per-connection slot ${slot} remains after its removal (chmod -R u+rwX, rm -rf); remove it by hand" >&2
        return 1
    }
}

# @description Refuse (exit 2) with the reason on stderr; in a connection the same line goes to the client first, and
#   a slot this connection made is removed (slot_remove) while no reaper has been forked, a slot that remains naming
#   itself. After the fork the slot is the reaper's (regime-cutover U343o): this process's exit closes the control
#   pipe, and the reaper removes the slot on that EOF.
# @arg $@ string The reason.
refuse() {
    if ((connection)); then printf 'acp-broker: refused: %s\n' "$*"; fi
    echo "acp-broker: refused: $*" >&2
    if [[ -z ${reaper} ]]; then
        if ! slot_remove; then :; fi
    fi
    exit 2
}

# @description The handler's one trap, armed for the making of the slot and the reaper's start alone (regime-cutover
#   U343p, verdict U343o-c5c2b968 P1 :1648, class handler-window, form remove): a hangup, interrupt or termination
#   from right before the slot's `mkdir` to right after the reaper's ready line is read removes the slot and exits 2,
#   one line on stderr and nothing on the stream; it waits on, signals and inspects neither omp nor the reaper nor
#   the record nor the report (U343g's trap waited on omp). What it removes follows the state the slot is in, not the
#   flag alone (regime-cutover U343q, verdict U343p-901e096a P1 :1669: a trap runs right after the simple command
#   the signal arrived in, so a TERM between the `mkdir` and `slot_made=1` ran slot_remove with the flag at 0 and
#   the fresh slot stood). With slot_made at 1 the slot may hold anything this connection made (the FIFOs), and
#   slot_remove takes it (idempotent, so a reaper already forked may remove it too; a slot that remains names
#   itself). With slot_made at 0 the only state the slot can have is the one a bare `mkdir` leaves, an empty
#   directory of this user, and `/bin/rmdir` takes exactly that and nothing else: not a non-empty directory, not a
#   file, not a link (it follows none), so a pre-existing entry at the name is never taken; its failure (nothing
#   made, or an entry that is not this connection's) is ignored. The stderr line names which removal happened.
# shellcheck disable=SC2329 # invoked by the trap armed before the slot's mkdir
slot_trap() {
    trap - HUP INT TERM
    if ((slot_made)); then
        if slot_remove; then
            echo "acp-broker: the connection ended by a signal before omp started; the slot removed" >&2
        fi
    else
        if /bin/rmdir -- "${slot}" 2> /dev/null; then
            echo "acp-broker: the connection ended by a signal as the slot was made; the empty slot removed" >&2
        fi
    fi
    exit 2
}

# @description Succeed when `$1` is a clean absolute path that srt cannot read as a pattern: `/` followed by

# @description Print the physical form of the absolute path `$1`: every symlink followed (at most 40 hops in all) and
#   the directory in its `pwd -P` form; a path that does not exist yet is its nearest existing ancestor's physical
#   form with the rest appended (the resolver of `hcom-seat`). Only shell builtins and `/usr/bin/readlink` run.
# @arg $1 string An absolute path.
# @stdout The physical path.
# @exitcode 1 Not absolute, a control character in it or in a link, more than 40 links, or a directory that cannot
#   be resolved.
physical_path() {
    local p=$1 tail='' link dir n=0
    [[ ${p} == /* && ${p} != *[[:cntrl:]]* ]] || return 1 # lint-comparisons: allow an absolute-path and content test
    while :; do
        while [[ ! -e ${p} && ! -L ${p} ]]; do
            tail=/${p##*/}${tail}
            p=${p%/*}
            [[ -n ${p} ]] || p=/
        done
        [[ -L ${p} ]] || break
        n=$((n + 1))
        ((n <= 40)) || return 1
        link=$(/usr/bin/readlink "${p}" && printf .) || return 1
        link=${link%.}
        link=${link%$'\n'}
        [[ -n ${link} && ${link} != *[[:cntrl:]]* ]] || return 1
        case ${link} in
        /*) p=${link} ;;
        *) p=${p%/*}/${link} ;;
        esac
    done
    if [[ ${p} == / ]]; then # lint-comparisons: allow the root itself, whose `cd //` form pwd prints as `//`
        printf '%s\n' "${tail:-/}"
    elif [[ -d ${p} ]]; then
        dir=$(cd "${p}/" && pwd -P) || return 1
        printf '%s%s\n' "${dir%/}" "${tail}"
    else
        dir=$(cd "${p%/*}/" && pwd -P) || return 1
        printf '%s/%s%s\n' "${dir%/}" "${p##*/}" "${tail}"
    fi
}

# @description Set the variable named `$2` to `$1` with A-Z made a-z (regime-cutover U95), by builtins only (a case
#   table, no `tr` and no locale; bash 3.2 has no `${x,,}`), so no producer's output is read back.
# @arg $1 string ASCII text.
# @arg $2 string The name of the variable to set.
# @exitcode 1 The variable cannot be set.
ascii_lower() {
    # The locals carry a prefix, so the caller's variable `$2` is never one of them (printf -v sets the nearest).
    local LC_ALL=C al_text=$1 al_out='' al_c al_before al_upper=ABCDEFGHIJKLMNOPQRSTUVWXYZ
    local al_lower=abcdefghijklmnopqrstuvwxyz
    [[ $2 != al_* && $2 != LC_ALL ]] || return 1
    while [[ -n ${al_text} ]]; do
        al_c=${al_text:0:1}
        al_text=${al_text:1}
        case ${al_c} in
        [ABCDEFGHIJKLMNOPQRSTUVWXYZ])
            al_before=${al_upper%%"${al_c}"*}
            al_c=${al_lower:${#al_before}:1}
            ;;
        esac
        al_out+=${al_c}
    done
    printf -v "$2" '%s' "${al_out}"
}

            [[ -n ${d} ]] || d=/
        done
        if [[ -e ${p} ]]; then tbl_exists+=(1); else tbl_exists+=(0); fi
        ancestors_of "${d}"
        all+=("${ancestors[@]}")
        counts+=("${#ancestors[@]}")
        rest=${rest}/
        ascii_lower "${rest}" folded || refuse "cannot fold the absent part of ${p}"
        [[ ${#folded} == "${#rest}" && ${folded} =~ ^(/[a-z0-9._@+-]+)*/$ ]] ||
            refuse "cannot fold the absent part of ${p} ('${folded}')"
        tbl_rest+=("${folded}")
    done
    (($#)) || return 0
    ids=$(printf '%s\0' "${all[@]}" | /usr/bin/xargs -0 /usr/bin/stat "${stat_args[@]}" --) ||
        refuse "cannot read the identity of a settings row or a directory above it"
    split_lines "${ids}"
    for line in ${lines[@]+"${lines[@]}"}; do
        [[ ${line} =~ ^[0-9]+:[0-9]+$ ]] || refuse "cannot read the identity of a settings row ('${line}')"
        ((i < ${#counts[@]})) || refuse 'more identities than settings rows were read'
        if ((k == 0)); then tbl_first+=("${line}"); fi
        chain+=${line}$'\n'
        k=$((k + 1))
        if ((k == counts[i])); then
            tbl_chain+=($'\n'"${chain}")
            chain='' k=0 i=$((i + 1))
        fi
    done
    ((i == $# && k == 0)) || refuse "cannot read the identity of every settings row (${i} of $#)"
    ((${#tbl_exists[@]} == $# && ${#tbl_first[@]} == $# && ${#tbl_chain[@]} == $# && ${#tbl_rest[@]} == $#)) ||
        refuse 'the identity table does not hold one entry per settings row'
}

# @description Decide from the identity table whether path `$2` is path `$1` or lies under it (regime-cutover U85,
#   U95), by identity for everything that exists, so a case alias on a case-insensitive volume is the path it aliases.
#   When `$1` exists: yes iff `$1`'s identity is in `$2`'s chain (`$2`'s deepest existing ancestor or a directory
#   above it). When `$1` does not exist, an existing `$2` cannot lie under it (no). When neither exists, `$2` can lie
#   under `$1` only if both have the same deepest existing ancestor; below it nothing exists, so no link or alias can
#   intervene, and the absent remainders decide: yes iff `$1`'s is `$2`'s or a prefix of it on a component boundary
#   (`a/b` is no prefix of `a/bc`), compared ASCII case-folded, since on a case-insensitive volume two spellings that
#   differ by case become one directory once made. No process runs.
# @arg $1 int The table index of the outer path.
# @arg $2 int The table index of the inner path.
# @exitcode 0 `$2` is `$1` or lies under it.
# @exitcode 1 It does not.
within_ix() {
    if ((tbl_exists[$1])); then
        [[ ${tbl_chain[$2]} == *$'\n'"${tbl_first[$1]}"$'\n'* ]] # lint-comparisons: allow compares (device, inode) identities, not paths
        return
    fi
    ((! tbl_exists[$2])) || return 1
    [[ ${tbl_first[$1]} == "${tbl_first[$2]}" ]] || return 1 # lint-comparisons: allow compares (device, inode) identities, not paths
    [[ ${tbl_rest[$2]} == "${tbl_rest[$1]}"* ]]              # lint-comparisons: allow absent components below one ancestor of the same identity, compared on component boundaries
}

# @description Set `phys_list` to the physical form of each absolute path `$@`, in order (regime-cutover U95), or
#   refuse (exit 2), in one subshell for all of them, present or absent (physical_path). Fail-closed per record: the
#   subshell writes exactly one record per path, the physical path or `fail`, and fails when a record cannot be
#   written; each record is checked against the clean-path grammar and assigned to its own index. Anything else, a
#   record too many or too few, or an index left without its result refuses.
# @arg $@ string Absolute paths.
# @set phys_list The physical paths, one per input.
physical_each() {
    local out line i=0 args=("$@")
    phys_list=()
    (($#)) || return 0
    out=$(for p in "$@"; do
        physical_path "${p}" || printf 'fail\n' || exit 1
    done) || refuse 'cannot resolve the settings rows (the resolving subshell failed)'
    split_lines "${out}"
    for line in ${lines[@]+"${lines[@]}"}; do
        ((i < $#)) || refuse 'cannot resolve the settings rows (more records than rows)'
        [[ ${line} != fail ]] || refuse "cannot resolve the settings row ${args[i]}"
        clean_path "${line}" || refuse "the settings row ${args[i]} resolves to '${line}', not a clean absolute path"
        phys_list[i]=${line}
        i=$((i + 1))
    done
    ((i == $# && ${#phys_list[@]} == $#)) || refuse "cannot resolve every settings row (${#phys_list[@]} of $#)"
}

# @description Judge the worker's grants against its denies (regime-cutover U343c; the settings_judge of `hcom-seat`,
#   U272f and U272g, whose rows the worker's are): every row of `allow_read` and `allow_write` against every row of
#   `deny_read` and `deny_write` (the macOS write patterns are not rows), and nothing else is the boundary. Every row
#   is resolved to its physical form first (physical_each: a link at the end of a row hides its ancestry from the
#   identity table), the table is filled once (identity_table), and a grant that is, contains or lies inside a deny
#   (within_ix either way) is a hit unless the pair is structural: both rows carry the same tag and the grant lies
#   where the role spells it (its physical path is the path in `allow_read_spelled` or `allow_write_spelled`). The
#   tags are the dsh row's, assigned by construction since this lever reads the deny table instead of rendering it:
#   the grants carry `slot` for the task cwd and the shared Claude Code directory and `profile` for the omp profile
#   (`allow_read_tags`, `allow_write_tags`); a denyRead row carries `slot` when it has the identity of the seats root
#   or of `~/.local/state/omp-seats` and `profile` when it has the identity of `~/.omp/profiles` (the three
#   references resolved and tabled with the rows); a denyWrite row carries `slot` when it lies strictly inside a
#   slot-tagged grant (the project-file rows under the task cwd, the configuration rows under the shared directory).
#   A profile that is a link elsewhere is spelled `~/.omp/profiles/dsh` and resolves elsewhere, so its tag exempts
#   nothing and it is judged like any row. The read grants go first, so the first hit names the row as the role
#   spells it.
# @set judge_grant The first grant that hits: `<kind> row <spelled path>` and, when it differs, its physical path in
#   parentheses; empty when every grant is clear.
# @set judge_deny The deny it hits: `<path>[ (<physical>)], a row of its <kind> list`; empty when clear.
# @exitcode 0 Every grant keeps clear of every deny it is not structurally paired with.
# @exitcode 1 A pair overlaps; `judge_grant` and `judge_deny` name it.
settings_judge() {
    local i j k n m rows=() kinds=() spelled=() tags=() refs=() ref_tags=(slot slot profile)
    for ((i = 0; i < ${#allow_read[@]}; i++)); do
        rows+=("${allow_read[i]}") kinds+=(allowRead) spelled+=("${allow_read_spelled[i]}") tags+=("${allow_read_tags[i]}")
    done
    for ((i = 0; i < ${#allow_write[@]}; i++)); do
        rows+=("${allow_write[i]}") kinds+=(allowWrite) spelled+=("${allow_write_spelled[i]}") tags+=("${allow_write_tags[i]}")
    done
    n=${#rows[@]}
    for ((i = 0; i < ${#deny_read[@]}; i++)); do
        rows+=("${deny_read[i]}") kinds+=(denyRead) spelled+=("${deny_read[i]}") tags+=('')
    done
    for ((i = 0; i < ${#deny_write[@]}; i++)); do
        [[ ${deny_write[i]} != *\** ]] || continue # lint-comparisons: allow a pattern test on one row, no path compare
        rows+=("${deny_write[i]}") kinds+=(denyWrite) spelled+=("${deny_write[i]}") tags+=('')
    done
    m=$((${#rows[@]} - n))
    judge_grant='' judge_deny=''
    ((n && m)) || return 0
    # The structural references take the table positions after the rows: the seats root, the default seats parent
    # and the profiles directory, each tagged as hcom-seat tags the deny row it renders there.
    refs=("${seats_real}" "${h}/.local/state/omp-seats" "${h}/.omp/profiles")
    physical_each "${rows[@]}" "${refs[@]}"
    identity_table "${phys_list[@]}"
    for ((j = n; j < n + m; j++)); do
        case ${kinds[j]} in
        denyRead)
            for ((k = 0; k < ${#refs[@]}; k++)); do
                i=$((n + m + k))
                if within_ix "${i}" "${j}" && within_ix "${j}" "${i}"; then tags[j]=${ref_tags[k]}; fi
            done
            ;;
        *)
            for ((i = 0; i < n; i++)); do
                [[ ${tags[i]} == slot ]] || continue # lint-comparisons: allow a tag word, no path compare
                if within_ix "${i}" "${j}" && ! within_ix "${j}" "${i}"; then tags[j]=slot; fi
            done
            ;;
        esac
    done
    for ((i = 0; i < n; i++)); do
        for ((j = n; j < n + m; j++)); do
            within_ix "${j}" "${i}" || within_ix "${i}" "${j}" || continue
            if [[ -n ${tags[i]} && ${tags[i]} == "${tags[j]}" && ${phys_list[i]} == "${spelled[i]}" ]]; then continue; fi # lint-comparisons: allow two tag words, and one row's spelled and physical forms from the same physical producers; the overlap was decided by identity above
            judge_grant="${kinds[i]} row ${spelled[i]}"
            judge_deny=${rows[j]}
            # The same row in two spellings when the spelled one goes through a link; the reader needs both.
            [[ ${phys_list[i]} == "${spelled[i]}" ]] || judge_grant+=" (${phys_list[i]})" # lint-comparisons: allow one row's spelled and physical forms, both from physical producers, no decision rests on it
            [[ ${phys_list[j]} == "${rows[j]}" ]] || judge_deny+=" (${phys_list[j]})"     # lint-comparisons: allow one row's rendered and physical forms, both from physical producers, no decision rests on it
            judge_deny+=", a row of its ${kinds[j]} list"
            return 1
        done
    done
}

# @description Set `rows` to the entries of the Router settings' `filesystem.$1` list, read whole by the system's
#   reader (`/usr/bin/plutil … json` on macOS, which escapes `/` as `\/`; `/usr/bin/jq -c` on Linux) and parsed here
#   from the one JSON array text: each element a double-quoted string holding no quote and no backslash once `\/` is
#   undone, each then a clean row (clean_row). Fail-closed: a failed read, a text that is no array of such strings,
#   or a row that is not clean refuses (exit 2) naming the file and the list.
# @arg $1 string The list: denyRead or denyWrite.
# @set rows The entries, in the file's order.
template_rows() {
    local list=$1 text rest item patterns=0 slash_esc='\/'
    [[ ${list} != denyWrite ]] || patterns=1
    case ${os} in
    Darwin) text=$(/usr/bin/plutil -extract "filesystem.${list}" json -o - "${template}") || refuse "cannot read filesystem.${list} from the Router's settings ${template}" ;;
    *) text=$(/usr/bin/jq -c ".filesystem.${list}" "${template}") || refuse "cannot read filesystem.${list} from the Router's settings ${template}" ;;
    esac
    text=${text%$'\n'}
    [[ ${text} == \[*\] ]] || refuse "filesystem.${list} in the Router's settings ${template} is not a JSON array ('${text}')"
    rest=${text#\[}
    rest=${rest%\]}
    rows=()
    while [[ -n ${rest} ]]; do
        [[ ${rest} == \"* ]] || refuse "filesystem.${list} in the Router's settings ${template} holds an element that is no string ('${rest}')"
        rest=${rest#\"}
        [[ ${rest} == *\"* ]] || refuse "filesystem.${list} in the Router's settings ${template} holds an unterminated string"
        item=${rest%%\"*}
        rest=${rest#*\"}
        item=${item//"${slash_esc}"/\/}
        clean_row "${item}" "${patterns}" || refuse "filesystem.${list} in the Router's settings ${template} holds '${item}', not a clean absolute path"
        rows+=("${item}")
        case ${rest} in
        '') ;;
        ,\"*) rest=${rest#,} ;;
        *) refuse "filesystem.${list} in the Router's settings ${template} is malformed after '${item}'" ;;
        esac
    done
}

# @description Print the first executable `$1` in the private tool directory (hcom-seat's links), else on the caller's
#   PATH; the private directory first, so the tool hcom-seat checked on the Router's launch is the one a worker runs.
# @arg $1 string The command name.
# @stdout The path, symlinks not yet followed.
# @exitcode 1 None found.
find_tool() {
    local path_dir path_dirs=()
    if [[ -n ${tools_real} && -x ${tools_real}/$1 && ! -d ${tools_real}/$1 ]]; then
        printf '%s\n' "${tools_real}/$1"
        return 0
    fi
    IFS=: read -r -a path_dirs <<< "${caller_path}"
    for path_dir in ${path_dirs[@]+"${path_dirs[@]}"}; do
        [[ ${path_dir} == /* && ${path_dir} != *[[:cntrl:]]* ]] || continue # lint-comparisons: allow an absolute-path and content test
        [[ -x ${path_dir}/$1 && ! -d ${path_dir}/$1 ]] || continue
        printf '%s\n' "${path_dir}/$1"
        return 0
    done
    return 1
}

# @description Refuse (exit 2) a tool file that a seat could rewrite: one that is or lies under the seats root or
#   the Router's scratch cwd (both physical, prefix on a whole component), or a mise shim (named mise, or found in or
#   resolving to a `shims` directory, as scripts/orchestrator/ci-rerun.sh decides it), which reads the cwd's config.
# @arg $1 string A label for the refusal.
# @arg $2 string The path the tool was found at.
# @arg $3 string Its physical file.
tool_place_check() {
    local d
    [[ ${3}/ != "${seats_real}"/* && ${3}/ != "${scratch_real}"/* ]] || # lint-comparisons: allow both operands physical (physical_path); the trailing slash makes the prefix a whole component
        refuse "$1 $3 lies under the seats root or the Router's scratch cwd, where a seat could rewrite it"
    [[ ${3##*/} != [Mm][Ii][Ss][Ee] ]] || refuse "$1 $3 is mise itself" # lint-comparisons: allow a file-name test, not a comparison of two paths
    for d in "${2%/*}" "${3%/*}"; do
        d=$(physical_path "${d:-/}") || refuse "cannot resolve the directory of $1 $2"
        [[ ${d}/ != */[Ss][Hh][Ii][Mm][Ss]/* ]] || refuse "no $1 from a shims directory ('$2'): a mise shim picks its version from the cwd" # lint-comparisons: allow a directory-name test, not a comparison of two paths
    done
}

# @description Resolve the tool `$1` to the argv that runs it with no lookup left (hcom-seat pin_exec): its physical
#   file (find_tool, physical_path), preceded by the physical interpreter when it is a `#!` script (`#!/abs/interp`
#   or `#!/usr/bin/env <name>`, no arguments; the name found as a tool is; an interpreter that is itself a script is
#   refused). Each file passes tool_place_check. A failure exits 2 naming `$1`.
# @arg $1 string The tool's name.
# @set pinned The argv.
pin_tool() {
    local label=$1 found file first magic interp name extra interp_path
    found=$(find_tool "$1") || refuse "${label} not found in the private tool directory ${tools_dir} or on the caller's PATH; run hcom-seat once (make update)"
    file=$(physical_path "${found}") || refuse "cannot resolve ${label} to a physical file: ${found}"
    [[ -f ${file} && -x ${file} && -r ${file} ]] || refuse "${label} ${found} resolves to ${file}, not a readable executable file"
    tool_place_check "${label}" "${found}" "${file}"
    # Each read's status is checked (pipefail covers head; /usr/bin/od only formats): a failed read is refused, never
    # taken for a file without an interpreter line.
    magic=$(/usr/bin/head -c 2 -- "${file}" | /usr/bin/od -An -tx1) || refuse "cannot read ${label}: ${file}"
    first=''
    if [[ ${magic//[[:space:]]/} == 2321 ]]; then # lint-comparisons: allow the hex bytes of `#!`, no path compare
        first=$(/usr/bin/head -n 1 -- "${file}") || refuse "cannot read the interpreter line of ${label}: ${file}"
    fi
    pinned=("${file}")
    case ${first} in
    '#!'*)
        extra=''
        read -r interp name extra <<< "${first#??}"
        if [[ ${interp} == /usr/bin/env ]]; then # lint-comparisons: allow the literal interpreter line, no path compare
            [[ -n ${name} && -z ${extra} && ${name} != -* ]] ||
                refuse "${label}: unsupported interpreter line '${first}' in ${file}"
            interp_path=$(find_tool "${name}") || refuse "${label}: interpreter ${name} not found in the private tool directory or on the caller's PATH"
        else
            [[ -z ${name} ]] || refuse "${label}: unsupported interpreter line '${first}' in ${file}"
            [[ ${interp} == /* ]] || # lint-comparisons: allow an absolute-path test
                refuse "${label}: interpreter line '${first}' in ${file} names no absolute path"
            interp_path=${interp}
        fi
        interp=$(physical_path "${interp_path}") || refuse "cannot resolve the interpreter of ${label}: ${interp_path}"
        [[ -f ${interp} && -x ${interp} && -r ${interp} ]] || refuse "the interpreter of ${label}, ${interp_path}, resolves to ${interp}, not a readable executable file"
        tool_place_check "the interpreter of ${label}" "${interp_path}" "${interp}"
        magic=$(/usr/bin/head -c 2 -- "${interp}" | /usr/bin/od -An -tx1) || refuse "cannot read the interpreter of ${label}: ${interp}"
        [[ ${magic//[[:space:]]/} != 2321 ]] || refuse "${label}: its interpreter ${interp} is itself a script" # lint-comparisons: allow the hex bytes of `#!`, no path compare
        pinned=("${interp}" "${file}")
        ;;
    esac
}

# @description The listener's signal trap: a hangup, interrupt or termination ends the relay (and, through it, every
#   worker it started; the relay removes its socket), removes the records once the socket is gone and then ends this
#   process by the same signal, its disposition restored; a socket that stands is exit 2 whatever the signal, nothing
#   removed.
# @arg $1 string The signal name.
# shellcheck disable=SC2329 # invoked by the trap installed below
broker_signal() {
    local removed=0
    trap - HUP INT TERM
    if [[ -n ${child} ]]; then
        if kill -TERM "${child}" 2> /dev/null; then :; fi
        if wait "${child}"; then :; fi
    fi
    broker_cleanup || removed=$?
    ((removed == 0)) || {
        echo "acp-broker: ended by SIG$1; the broker exits 2 because the socket stands and nothing was removed" >&2
        exit 2
    }
    kill -s "$1" $$
    exit 2
}

# @description Wait for the child `$1` and set `child_status` to its status, reading `wait` again when a trapped
#   signal interrupted it (bash returns from `wait` with the signal's status and runs the trap first).
# @arg $1 int The pid.
# @set child_status The exit status.
wait_child() {

    while :; do
        # The status is read in the else branch, where `$?` is still `wait`'s own.
        if wait "$1"; then
            child_status=0
        else
            child_status=$?
        fi
        if kill -0 "$1" 2> /dev/null; then continue; fi
        return 0
    done
}

version=$(/usr/bin/env -i "${worker_env[@]}" "${srt_cmd[@]}" --settings "${settings}" -- /usr/bin/env PATH="${seat_path}" "${omp_cmd[@]}" --version < /dev/null 4>&-) ||
    refuse "cannot read the omp version (${omp_cmd[*]} --version failed under ${srt_cmd[*]}); omp acp not started"
version=${version%%$'\n'*}
version=${version%% *}
# omp 18.12.6 prints `omp/<version>`; the log carries the version alone.
version=${version#omp/}
[[ ${version} =~ ^[0-9A-Za-z.+-]+$ ]] || refuse "cannot read the omp version from '${version}'; omp acp not started"
log_line "omp acp start omp ${version} cwd ${task} repo ${repo} ref ${ref} base ${base}"
# The one broker line on the stream, before omp starts; everything after it is the worker's. The reaper has been up
# since the slot was made (its ready line read above), so from here on it owns the exit report too, and this process
# refuses nothing more on the stream once the record line is written.
printf 'acp-broker: record %s\n' "${slot_name}" || refuse "cannot write the record line to the connection; omp acp not started"
# omp runs as a child, not an exec, so the push can follow and the reaper be told; the explicit `<&0` keeps the
# connection as its stdin (a background command gets /dev/null otherwise), its stdout is the connection, and the
# control pipe is closed for it. Job control is on for the one start, so the job gets a process group of its own and
# the reaper can end omp with every descendant in that group by one signal (verdict U343d-136fbe0a P2 :1079).
set -m
/usr/bin/env -i "${worker_env[@]}" "${srt_cmd[@]}" --settings "${settings}" -- /usr/bin/env PATH="${seat_path}" "${omp_cmd[@]}" acp --profile dsh <&0 4>&- &
child=$!
set +m
printf 'start %s\n' "${child}" >&4 || echo "acp-broker: cannot tell the reaper that omp started (pid ${child})" >&2
wait_child "${child}"
status=${child_status}
child=''
printf 'exited %s\n' "${status}" >&4 || echo "acp-broker: cannot tell the reaper that omp exited ${status}" >&2
# The push, then the `done` line: the reaper writes the exit line, removes the slot and writes the record's final
# line (status 2 when the slot remains; verdict U343d-136fbe0a P2 :1157), and its exit closes the connection's last
# copy of stdout. A signal that ends this process anywhere above closes the pipe instead, with the same teardown.
push_back
printf 'done %s %s\n' "${status}" "${push_word}" >&4 || echo "acp-broker: cannot tell the reaper that the push ended (${push_word})" >&2
exec 4>&-
if wait "${reaper}"; then
    reaper_status=0
else
    reaper_status=$?
fi
((reaper_status == 0)) || {
    echo "acp-broker: omp ended ${status}; the connection exits 2 because the reaper (pid ${reaper}) ended ${reaper_status} (a per-connection slot or the record remains)" >&2
    exit 2
}
exit "${status}"
