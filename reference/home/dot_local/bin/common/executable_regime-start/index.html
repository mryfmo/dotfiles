    home/dot_local/bin/common/executable_regime-start¶

    Regime-cutover U36: the operator starts one omp seat and talks to it; that seat starts the others (the hcom-orchestration skill's step 0). Run regime-start [<program>] in the repository checkout. The store and team are an explicit HCOM_DIR and HCOM_TEAM when set; else, for the dotfiles program before its one-time store move (slug dotfiles, no ~/.local/state/hcom/dotfiles/origin, and ~/.local/state/hcom-regime/hcom.db present), the pre-move pair ~/.local/state/hcom-regime and regime, because a seat's environment carries no HCOM_TEAM and the slug would name another thread (decisions 2026-10-24T10:19:06Z); else the sourced regime-env derives both (U33). An explicit store must already exist. When a seat tagged orchestrator is alive in the store (hcom status active, listening or blocked) it is reported, not duplicated; otherwise hcom-seat orchestrator (the copy beside this file) launches it from the checkout, and the command waits up to 120 s for it to be alive. Either way the checkout is registered for the supervisor (toplevel, store, team, program, last seat name), and the command prints the seat's name and where to type. A checkout already registered on another store or team is refused until regime-stop, and a store whose origin names another checkout is refused, an explicit one included.

    #!/usr/bin/env bash
# @file regime-start
# @brief Start a program's orchestrator seat, or, as the supervisor's pass, relaunch every registered one that died.
# @description
#   Regime-cutover U36: the operator starts one omp seat and talks to it; that seat starts the others (the
#   hcom-orchestration skill's step 0). Run `regime-start [<program>]` in the repository checkout. The store and
#   team are an explicit HCOM_DIR and HCOM_TEAM when set; else, for the dotfiles program before its one-time store
#   move (slug `dotfiles`, no `~/.local/state/hcom/dotfiles/origin`, and `~/.local/state/hcom-regime/hcom.db`
#   present), the pre-move pair `~/.local/state/hcom-regime` and `regime`, because a seat's environment carries
#   no HCOM_TEAM and the slug would name another thread (decisions 2026-10-24T10:19:06Z); else the sourced
#   regime-env derives both (U33). An explicit store must already exist. When a seat tagged `orchestrator` is
#   alive in the store (hcom status active, listening or blocked) it is reported, not duplicated; otherwise
#   `hcom-seat orchestrator` (the copy beside this file) launches it from the checkout, and the command waits up
#   to 120 s for it to be alive. Either way the checkout is registered for the supervisor (toplevel, store, team,
#   program, last seat name), and the command prints the seat's name and where to type. A checkout already
#   registered on another store or team is refused until regime-stop, and a store whose `origin` names another
#   checkout is refused, an explicit one included.
#
#   `--supervise` is one supervisor pass, run every 60 s by the launchd agent
#   `com.mryfmo.dotfiles.regime-supervisor` (macOS) or the `regime-supervisor.timer` user unit (Linux) once the
#   operator enables them; both ship disabled. For each registration with no orchestrator alive in its store it reads
#   hcom's newest stop record of the last seat (the `(latest)` event of `hcom list --stopped <base name>`):
#   `killed by:cli`, an operator's `hcom kill`, ends the supervision (the registration is removed); a record that
#   cannot be read launches nothing; anything else (a crash, a herdr restart) is relaunched like the interactive
#   form, at most once per backoff window (60 s doubling, saturated at 30 minutes, reset by a seat seen alive). Every
#   decision is a line in `<store>/supervisor.log`.
#
#   `--stop [--all]` is regime-stop (the lever beside this file runs it). Start, stop and the supervisor pass take
#   one host lock, the kernel's flock on `~/.local/state/regime-supervisor/supervisor.lock` (take_lock), so a stop
#   never interleaves with a relaunch: start and stop wait for it, a pass that finds it held skips. A registration is
#   `~/.local/state/regime-supervisor/<slug>-<first 12 hex of sha256(toplevel)>.conf`, one per checkout, with every
#   key exactly once (toplevel, hcom_dir, hcom_team, program, seat, base, fails, next); a registration that breaks
#   that, names another toplevel, or whose store's `origin` names another checkout is never launched, replaced or
#   stopped from.
# @arg $1 string Program name (`^[a-z0-9][a-z0-9_-]{0,63}$`) whose store `.orchestration/orchestrate/<program>/` must exist; optional.
# @option --supervise One supervisor pass over every registration; no other argument.
# @option --stop [--all] regime-stop.
# @env HCOM_DIR The hcom store (default: as described above).
# @env HCOM_TEAM The team (default: as described above).
# @stdout The seat's name and where to type (interactive form).
# @exitcode 0 The orchestrator is alive, or the pass ran.
# @exitcode 1 The launch failed or the seat did not come alive within 120 s.
# @exitcode 2 A refusal, before anything is started.
# @example
#   regime-start regime-cutover
set -euo pipefail

# @description Refuse to run unless this file's sha256 equals its pin in the manifest beside it (regime-cutover U31):
#   `~/.agents/agent-config.yaml` for the installed copy under ~/.local/bin/common, which a sandboxed session cannot
#   write, else the checkout's home/dot_agents/agent-config.yaml. Every read is checked; a failure exits 2 naming the
#   file. The pin is the `claude.levers.sha256.<name>` line, which `scripts/generate-agent-configs.py` refreshes.
# @set LEVER_MANIFEST The manifest the pin came from.
# @set LEVER_DIR The physical directory of this file, where the sourced regime-env helper sits (U33).
lever_self_check() {
    local src dir self name pin digest home
    # The directory is split from the path in the shell, so no external producer (dirname) can fail unseen.
    src=${BASH_SOURCE[0]}
    case ${src} in
    */*) dir=${src%/*} ;;
    *) dir=. ;;
    esac
    [[ -n ${dir} ]] || dir=/
    dir=$(cd "${dir}" && pwd -P) || {
        echo "${src##*/}: refused: cannot resolve the script directory of ${src}" >&2
        exit 2
    }
    self=${dir}/${BASH_SOURCE[0]##*/}
    LEVER_DIR=${dir}
    name=${self##*/}
    home=$(cd "${HOME}" 2> /dev/null && pwd -P) || home=''
    if [[ -n ${home} && ${dir} == "${home}/.local/bin/common" ]]; then # lint-comparisons: allow both sides are pwd -P forms
        LEVER_MANIFEST=${HOME}/.agents/agent-config.yaml
    else
        LEVER_MANIFEST=${dir}/../../home/dot_agents/agent-config.yaml
    fi
    [[ -r ${LEVER_MANIFEST} ]] || {
        echo "${name}: refused: manifest unreadable: ${LEVER_MANIFEST}" >&2
        exit 2
    }
    pin=$(sed -n -E "s/^      ${name//./\\.}: ([0-9a-f]{64})\$/\\1/p" "${LEVER_MANIFEST}") || {
        echo "${name}: refused: cannot read ${LEVER_MANIFEST}" >&2
        exit 2
    }
    [[ ${pin} =~ ^[0-9a-f]{64}$ ]] || {
        echo "${name}: refused: no single sha256 pin for ${name} in ${LEVER_MANIFEST}" >&2
        exit 2
    }
    if command -v sha256sum > /dev/null; then
        digest=$(sha256sum < "${self}") || digest=''
    else
        digest=$(shasum -a 256 < "${self}") || digest=''
    fi
    digest=${digest%% *}
    [[ ${digest} =~ ^[0-9a-f]{64}$ ]] || {
        echo "${name}: refused: cannot hash ${self}" >&2
        exit 2
    }
    [[ ${digest} == "${pin}" ]] || {
        echo "${name}: refused: sha256 ${digest} of ${self} differs from the pin in ${LEVER_MANIFEST}" >&2
        exit 2
    }
}
lever_self_check

usage='usage: regime-start [<program>] | regime-start --supervise | regime-start --stop [--all]'
me=regime-start
registry=${HOME}/.local/state/regime-supervisor
lock=${registry}/supervisor.lock
lock_holder=''
name_grammar='^[a-z0-9][a-z0-9_-]{0,63}$'
wait_seconds=${REGIME_START_WAIT_SECONDS:-120}
[[ ${wait_seconds} =~ ^[0-9]{1,4}$ ]] || wait_seconds=120
lock_wait=${REGIME_LOCK_WAIT_SECONDS:-300}
[[ ${lock_wait} =~ ^[0-9]{1,4}$ ]] || lock_wait=300

refuse() {
    echo "${me}: $*" >&2
    exit 2
}

# @description Take the host's one regime lock, which regime-start, its supervisor pass and regime-stop share, so a
#   stop never interleaves with a launch (U36 F2). The lock is the kernel's flock(2) on the file `lock`, held through
#   descriptor 9 of this process: perl (on both hosts; macOS ships no flock command) takes it on that inherited
#   descriptor and exits, and the lock stays with the open file, which this process keeps, and the subshell that runs
#   a launch keeps until the launch returns. Every hcom call and hcom-seat get the descriptor closed, so no seat or
#   hcom process ever holds it. The kernel releases it when the last holder exits or dies, so nothing is ever
#   recovered by age or pid, no other process can release it, and the file itself is never removed. The holder
#   writes its pid into the file, for messages only. A lock file replaced after it was opened (another inode at the
#   path) is opened again, at most three times.
# @arg $1 string `wait` (up to REGIME_LOCK_WAIT_SECONDS, default 300) or `try` (once).
# @set lock_holder The pid the current holder wrote, when the lock is held.
# @exitcode 1 The lock is held by another process.
take_lock() {
    local seconds=0 status attempt=0
    [[ $1 == wait ]] && seconds=${lock_wait}
    mkdir -p "${registry}" || refuse "cannot create ${registry}"
    while ((attempt < 3)); do
        attempt=$((attempt + 1))
        [[ ! -L ${lock} ]] || refuse "${lock} is a symlink; nothing changed"
        exec 9>> "${lock}" || refuse "cannot open ${lock}; nothing changed"
        status=0
        perl -MFcntl=:flock -e '
            my ($wait, $owner, $path) = @ARGV;
            open(my $fh, ">>&=", 9) or exit 3;
            for my $try (0 .. $wait) {
                if (flock($fh, LOCK_EX | LOCK_NB)) {
                    my @held = stat($fh);
                    my @named = stat($path);
                    exit 4 unless @named && $named[0] == $held[0] && $named[1] == $held[1];
                    truncate($fh, 0) or exit 3;
                    syswrite($fh, "$owner\n") or exit 3;
                    exit 0;
                }
                sleep 1 if $try < $wait;
            }
            exit 1;' "${seconds}" "$$" "${lock}" || status=$?
        case ${status} in
        0) return 0 ;;
        1)
            lock_holder=$(cat "${lock}" 2> /dev/null) || lock_holder=''
            [[ ${lock_holder} =~ ^[0-9]+$ ]] || lock_holder=unknown
            exec 9>&-
            return 1
            ;;
        4) exec 9>&- ;;
        *) refuse "cannot take the flock on ${lock} (perl exited ${status}); nothing changed" ;;
        esac
    done
    refuse "${lock} was replaced while it was being locked, three times; nothing changed"
}

# @description Print the registration file of a checkout: `<slug>-<first 12 hex of sha256(toplevel)>.conf`, so two
#   checkouts with one basename never share a registration (U36 F3).
# @arg $1 path The physical toplevel.
# @arg $2 string Its slug.
# @exitcode 1 The digest could not be computed.
registration_path() {
    local digest
    if command -v sha256sum > /dev/null; then
        digest=$(printf '%s' "$1" | sha256sum) || return 1
    else
        digest=$(printf '%s' "$1" | shasum -a 256) || return 1
    fi
    digest=${digest%% *}
    [[ ${digest} =~ ^[0-9a-f]{64}$ ]] || return 1
    printf '%s/%s-%s.conf\n' "${registry}" "$2" "${digest:0:12}"
}

# @description Read a registration into reg_top, reg_store, reg_team, reg_program, reg_seat, reg_base, reg_fails and
#   reg_next. Every key must appear exactly once and nothing else may (U36 F4); the seat is `orchestrator-<base>`;
#   the counters have bounded lengths (U36 F5); and the file must be the one registration_path names for its own
#   toplevel, so a registration copied to another checkout's name is refused, and a store with an `origin` file
#   (a derived one, written by regime-init) must name that toplevel (U36 F3).
# @arg $1 path The registration file.
# @stdout Nothing.
# @stderr Nothing.
# @set reg_error Why the registration is unusable, empty when it is usable.
read_registration() {
    local conf=$1 line key value expected slug recorded
    # Bash 3.2 (macOS /bin/bash) has no associative arrays, so the keys seen are one space-separated string.
    local seen=' '
    reg_error='' reg_top='' reg_store='' reg_team='' reg_program='' reg_seat='' reg_base='' reg_fails='' reg_next=''
    if ! [[ -f ${conf} && -r ${conf} ]]; then
        reg_error='unreadable'
        return 0
    fi
    while IFS= read -r line || [[ -n ${line} ]]; do
        key=${line%%=*}
        value=${line#*=}
        if [[ ${line} != *=* ]]; then
            reg_error="a line without a key: '${line}'"
            return 0
        fi
        case ${key} in
        toplevel | hcom_dir | hcom_team | program | seat | base | fails | next) ;;
        *)
            reg_error="unknown key '${key}'"
            return 0
            ;;
        esac
        if [[ ${seen} == *" ${key} "* ]]; then
            reg_error="key '${key}' appears twice"
            return 0
        fi
        seen+="${key} "
        printf -v "reg_${key/toplevel/top}" '%s' "${value}"
    done < "${conf}" || {
        reg_error='unreadable'
        return 0
    }
    reg_store=${reg_hcom_dir:-} reg_team=${reg_hcom_team:-}
    for key in toplevel hcom_dir hcom_team program seat base fails next; do
        if [[ ${seen} != *" ${key} "* ]]; then
            reg_error="key '${key}' is missing"
            return 0
        fi
    done
    if [[ ${reg_top} != /* || ! -d ${reg_top} || ${reg_store} != /* || ! -d ${reg_store} ]]; then # lint-comparisons: allow an absolute-path prefix test, not a comparison of two paths
        reg_error='the toplevel or the store is not an existing absolute directory'
    elif [[ ! ${reg_team} =~ ${name_grammar} || ! ${reg_base} =~ ${name_grammar} ]]; then
        reg_error='the team or the base name is outside the grammar'
    elif [[ -n ${reg_program} && ! ${reg_program} =~ ${name_grammar} ]]; then
        reg_error='the program is outside the grammar'
    elif [[ ${reg_seat} != "orchestrator-${reg_base}" ]]; then
        reg_error="the seat '${reg_seat}' is not orchestrator-<base>"
    elif [[ ! ${reg_fails} =~ ^[0-9]{1,4}$ || ! ${reg_next} =~ ^[0-9]{1,12}$ ]]; then
        reg_error='the counters are not bounded numbers'
    elif ! slug=$(regime_slug "${reg_top}" 2> /dev/null) || ! expected=$(registration_path "${reg_top}" "${slug}"); then
        reg_error="cannot derive the registration name of ${reg_top}"
    elif [[ ${conf##*/} != "${expected##*/}" ]]; then
        reg_error="the file name does not belong to the toplevel ${reg_top}"
    elif [[ -e ${reg_store}/origin ]]; then
        if ! recorded=$(cat "${reg_store}/origin" 2> /dev/null); then
            reg_error="cannot read ${reg_store}/origin"
        elif [[ ${recorded} != "${reg_top}" ]]; then # lint-comparisons: allow both are pwd -P forms (regime-init, regime_toplevel)
            reg_error="the store ${reg_store} belongs to ${recorded}, not ${reg_top}"
        fi
    fi
    return 0
}

# @description Append one timestamped line to the store's supervisor log; a log that cannot be written is said on stderr.
# @arg $1 path The hcom store.
# @arg $@ string The line.
log_line() {
    local store=$1 stamp
    shift
    stamp=$(date -u +%Y-%m-%dT%H:%M:%SZ) || stamp=unknown-time
    printf '%s %s\n' "${stamp}" "$*" >> "${store}/supervisor.log" 2> /dev/null ||
        echo "regime-start: cannot write ${store}/supervisor.log: $*" >&2
}

# @description Print `<name> <base name>` of every alive seat tagged orchestrator in the store HCOM_DIR names.
# @exitcode 1 hcom could not be read (a failed read is never "no seat").
alive_orchestrators() {
    local listing name base tag status
    listing=$(hcom list --format '{name} {base_name} {tag} {status}' 9>&-) || return 1
    while read -r name base tag status; do
        [[ ${tag} == orchestrator ]] || continue
        case ${status} in
        active | listening | blocked) printf '%s %s\n' "${name}" "${base}" ;;
        esac
    done <<< "${listing}"
}

# @description Launch the orchestrator seat from the checkout and wait for it to be alive.
# @arg $1 path The checkout's toplevel.
# @stdout `<name> <base name>` of the seat that came alive.
# @exitcode 1 hcom-seat failed, or no seat was alive within the wait.
launch_orchestrator() {
    local top=$1 status=0 waited=0 alive=''
    # Killing regime-start cannot release the lock while a launch runs (U36 F2): both callers run this function in a
    # command-substitution subshell, which keeps descriptor 9 until the launch and its wait return, and the subshell
    # below keeps it until hcom-seat returns, so the hold does not depend on the caller (the command after hcom-seat
    # keeps bash from exec-ing it in place of that subshell). hcom-seat, and the hcom and seat processes it starts,
    # get the descriptor closed, so a detached seat never holds the lock after the launch.
    (
        cd "${top}" || exit 1
        "${LEVER_DIR}/hcom-seat" orchestrator 9>&-
        launched=$?
        exit "${launched}"
    ) >&2 || status=$?
    # hcom's status 2 is "still launching": wait for the seat like a ready one.
    if ((status != 0 && status != 2)); then
        echo "regime-start: hcom-seat orchestrator exited ${status}" >&2
        return 1
    fi
    while ((waited <= wait_seconds)); do
        if alive=$(alive_orchestrators) && [[ -n ${alive} ]]; then
            printf '%s\n' "${alive%%$'\n'*}"
            return 0
        fi
        sleep 2
        waited=$((waited + 2))
    done
    echo "regime-start: no orchestrator seat alive in ${HCOM_DIR} after ${wait_seconds} s" >&2
    return 1
}

# @description Write a registration atomically (a temporary file, then a rename).
# @arg $1 path The registration file.
# @arg $@ string `key=value` lines.
write_registration() {
    local file=$1 temp
    shift
    mkdir -p "${registry}"
    temp=$(mktemp "${file}.XXXXXX")
    printf '%s\n' "$@" > "${temp}"
    mv -f "${temp}" "${file}"
}

# @description Print the backoff after the given number of failed launches: 60 s doubling, saturated at 1800 s
#   before it can overflow (U36 F5).
# @arg $1 int Failed launches in a row, at least 1.
backoff_seconds() {
    local delay=60 step=1
    while ((step < $1 && delay < 1800)); do
        delay=$((delay * 2))
        step=$((step + 1))
    done
    ((delay <= 1800)) || delay=1800
    printf '%s\n' "${delay}"
}

mode=start all=0
case ${1:-} in
--supervise)
    (($# == 1)) || refuse "${usage}"
    mode=supervise
    ;;
--stop)
    me=regime-stop
    case ${#}:${2:-} in
    1:) ;;
    2:--all) all=1 ;;
    *) refuse 'usage: regime-stop [--all]' ;;
    esac
    mode=stop
    ;;
*) (($# <= 1)) || refuse "${usage}" ;;
esac
command -v hcom > /dev/null || refuse 'hcom not found on PATH'
[[ -x ${LEVER_DIR}/hcom-seat ]] || refuse "hcom-seat is not beside this script: ${LEVER_DIR}/hcom-seat"
[[ -r ${LEVER_DIR}/regime-env ]] || refuse "regime-env unreadable beside this script: ${LEVER_DIR}/regime-env"
# shellcheck source=scripts/orchestrator/regime-env
. "${LEVER_DIR}/regime-env"

if [[ ${mode} == supervise ]]; then
    # A pass that finds the lock held skips: the next one comes in 60 s.
    if ! take_lock try; then
        echo "regime-start: pid ${lock_holder} holds ${lock}; this pass skips" >&2
        exit 0
    fi
    now=$(date +%s)
    for conf in "${registry}"/*.conf; do
        [[ -f ${conf} ]] || continue
        newest=''
        read_registration "${conf}"
        if [[ -n ${reg_error} ]]; then
            echo "regime-start: skipped ${conf}: ${reg_error}" >&2
            if [[ ${reg_store} == /* && -d ${reg_store} ]]; then # lint-comparisons: allow an absolute-path prefix test
                log_line "${reg_store}" "skipped ${conf}: ${reg_error}; nothing launched"
            fi
            continue
        fi
        top=${reg_top} store=${reg_store} team=${reg_team} program=${reg_program}
        seat=${reg_seat} base=${reg_base} fails=$((10#${reg_fails})) next=$((10#${reg_next}))
        export HCOM_DIR=${store} HCOM_TEAM=${team}
        if ! alive=$(alive_orchestrators); then
            log_line "${store}" "hcom list failed; nothing launched"
            continue
        fi
        if [[ -n ${alive} ]]; then
            read -r seat base <<< "${alive%%$'\n'*}"
            if ((fails != 0)) || [[ ${seat} != "${reg_seat}" ]]; then
                write_registration "${conf}" "toplevel=${top}" "hcom_dir=${store}" "hcom_team=${team}" "program=${program}" "seat=${seat}" "base=${base}" fails=0 next=0
            fi
            continue
        fi
        if ! stopped=$(hcom list --stopped "${base}" 9>&-); then
            log_line "${store}" "hcom list --stopped ${base} failed; nothing launched"
            continue
        fi
        # The named form ends with `Stop history (<n> events):` and one `<age>  [<reason> by:<who>]` line per event,
        # the newest marked `(latest)` (hcom 0.7.30 and 0.7.31, measured; 0.7.31 on 2026-10-24, U22b). A stop record
        # that cannot be read is no license.
        newest=$(grep -m1 -E '\[[^]]+\] \(latest\)$' <<< "${stopped}" || true)
        newest=${newest#"${newest%%[![:space:]]*}"}
        if [[ -z ${newest} ]]; then
            log_line "${store}" "cannot read the stop record of ${seat}; nothing launched"
            continue
        fi
        if [[ ${newest} == *"[killed by:cli] (latest)" ]]; then
            log_line "${store}" "${seat} was stopped with hcom kill; supervision of ${top} ends (registration removed)"
            rm -f "${conf}"
            continue
        fi
        if ((now < next)); then
            continue
        fi
        log_line "${store}" "no orchestrator alive (last seat ${seat}: ${newest}); relaunching from ${top}"
        if alive=$(launch_orchestrator "${top}" 2>> "${store}/supervisor.log"); then
            read -r seat base <<< "${alive}"
            log_line "${store}" "relaunched ${seat}"
            write_registration "${conf}" "toplevel=${top}" "hcom_dir=${store}" "hcom_team=${team}" "program=${program}" "seat=${seat}" "base=${base}" fails=0 next=0
        else
            ((fails >= 9999)) || fails=$((fails + 1))
            delay=$(backoff_seconds "${fails}")
            log_line "${store}" "relaunch failed (${fails} in a row); next attempt in ${delay} s"
            write_registration "${conf}" "toplevel=${top}" "hcom_dir=${store}" "hcom_team=${team}" "program=${program}" "seat=${seat}" "base=${base}" "fails=${fails}" "next=$((now + delay))"
        fi
    done
    exit 0
fi

# @description Choose the store and team of this checkout when it has no registration: an explicit pair, else the
#   dotfiles pre-move pair, else regime-env's derivation. A store whose `origin` names another checkout is refused,
#   an explicit one included (U36 F3).
choose_store() {
    local legacy=${HOME}/.local/state/hcom-regime recorded
    if [[ -z ${HCOM_DIR:-} && -z ${HCOM_TEAM:-} && ${slug} == dotfiles && ! -e ${HOME}/.local/state/hcom/dotfiles/origin && -f ${legacy}/hcom.db ]]; then
        # The dotfiles program before its one-time store move (README "Run a program in another repository").
        export HCOM_DIR=${legacy} HCOM_TEAM=regime
        echo "${me}: the dotfiles program has not moved its store yet; using HCOM_DIR=${HCOM_DIR} HCOM_TEAM=${HCOM_TEAM}" >&2
    fi
    regime_env "${me}" || exit 2
    [[ -d ${HCOM_DIR} ]] || refuse "the hcom store ${HCOM_DIR} does not exist; run regime-init <program> first or set an existing store"
    [[ ${HCOM_TEAM} =~ ${name_grammar} ]] || refuse "team '${HCOM_TEAM}' is outside ${name_grammar}"
    if [[ -e ${HCOM_DIR}/origin ]]; then
        recorded=$(cat "${HCOM_DIR}/origin") || refuse "cannot read ${HCOM_DIR}/origin"
        if [[ ${recorded} != "${top}" ]]; then # lint-comparisons: allow both are pwd -P forms (regime-init, regime_toplevel)
            refuse "the store ${HCOM_DIR} belongs to ${recorded}, not ${top}"
        fi
    fi
}

top=$(regime_toplevel) || refuse "run ${me} inside the repository checkout"
slug=$(regime_slug "${top}") || refuse "cannot derive a repository slug from ${top}"
conf=$(registration_path "${top}" "${slug}") || refuse "cannot hash the toplevel ${top}"

if [[ ${mode} == stop ]]; then
    take_lock wait || refuse "pid ${lock_holder} still holds ${lock} after ${lock_wait} s; nothing changed"
    if [[ -e ${conf} ]]; then
        # The registered store wins: it is the one the supervisor would relaunch into.
        read_registration "${conf}"
        [[ -z ${reg_error} ]] || refuse "the registration ${conf} is unusable (${reg_error}); check it and remove it by hand; nothing changed"
        if [[ ${reg_top} != "${top}" ]]; then # lint-comparisons: allow both are pwd -P forms
            refuse "the registration ${conf} names ${reg_top}, not ${top}; nothing changed"
        fi
        export HCOM_DIR=${reg_store} HCOM_TEAM=${reg_team}
    else
        choose_store
    fi
    listing=$(hcom list --format '{name} {tag} {status}' 9>&-) || refuse "cannot read the seats of ${HCOM_DIR} (hcom list failed); nothing changed"
    rm -f "${conf}"
    log_line "${HCOM_DIR}" "regime-stop: supervision of ${top} ended"
    failed=0 stopped=0
    while read -r name tag status; do
        [[ -n ${name} ]] || continue
        case ${status} in active | listening | blocked) ;; *) continue ;; esac
        ((all)) || [[ ${tag} == orchestrator ]] || continue
        if hcom kill "${name}" --go > /dev/null 9>&-; then
            echo "regime-stop: stopped ${name}"
            log_line "${HCOM_DIR}" "regime-stop: stopped ${name}"
            stopped=$((stopped + 1))
        else
            echo "regime-stop: hcom kill ${name} failed" >&2
            failed=1
        fi
    done <<< "${listing}"
    ((stopped || failed)) || echo "regime-stop: no seat to stop in ${HCOM_DIR}; supervision ended"
    exit "${failed}"
fi

program=${1:-}
[[ -z ${program} || ${program} =~ ${name_grammar} ]] || refuse "program '${program}' is outside ${name_grammar}"
choose_store
store_dir=''
if [[ -n ${program} ]]; then
    store_dir=${top}/.orchestration/orchestrate/${program}
    [[ -d ${store_dir} ]] || refuse "no program store ${store_dir}; run regime-init ${program} first"
fi
take_lock wait || refuse "pid ${lock_holder} still holds ${lock} after ${lock_wait} s; nothing changed"
# This checkout's own registration is replaced only for the same store and team (U36 F3).
if [[ -e ${conf} ]]; then
    read_registration "${conf}"
    [[ -z ${reg_error} ]] || refuse "the registration ${conf} is unusable (${reg_error}); check it, remove it, then run regime-start again"
    if [[ ${reg_store} != "${HCOM_DIR}" || ${reg_team} != "${HCOM_TEAM}" ]]; then # lint-comparisons: allow both are the absolute store paths regime-start wrote
        refuse "this checkout is registered on the store ${reg_store} and team ${reg_team}; run regime-stop first to move it to ${HCOM_DIR} and ${HCOM_TEAM}"
    fi
fi
alive=$(alive_orchestrators) || refuse "cannot read the seats of ${HCOM_DIR} (hcom list failed)"
if [[ -n ${alive} ]]; then
    read -r seat base <<< "${alive%%$'\n'*}"
    state="is already alive"
    log_line "${HCOM_DIR}" "regime-start found ${seat} alive; registered for the supervisor"
else
    alive=$(launch_orchestrator "${top}") || {
        log_line "${HCOM_DIR}" "regime-start could not launch the orchestrator from ${top}"
        exit 1
    }
    read -r seat base <<< "${alive}"
    state="is listening"
    log_line "${HCOM_DIR}" "regime-start launched ${seat} from ${top}"
fi
write_registration "${conf}" "toplevel=${top}" "hcom_dir=${HCOM_DIR}" "hcom_team=${HCOM_TEAM}" "program=${program}" "seat=${seat}" "base=${base}" fails=0 next=0
# shellcheck disable=SC2016 # the backticks are text: the pane's name, set off for the reader
printf '%s %s on team %s. Type your instruction in the pane `orchestrator`; the store is %s.\n' \
    "${seat}" "${state}" "${HCOM_TEAM}" "${store_dir:-${HCOM_DIR}}"
